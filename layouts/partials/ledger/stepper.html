{{- /* A feature's Alpha, Beta and GA steps. A released stage shows its release, month and the posts and talks on it. A stage it never had below its highest released one is "Skipped"; anything above that, a GA target included, is "Not yet". Input: a data/ledger.yaml feature. */ -}} {{- $f := . }} {{- $order := slice "alpha" "beta" "ga" }} {{- $names := dict "alpha" "Alpha" "beta" "Beta" "ga" "GA" }} {{- $done := dict }} {{- $top := -1 }} {{- range $f.stages }} {{- if .target }}{{ continue }}{{ end }} {{- $done = merge $done (dict .stage .) }} {{- end }} {{- range $i, $k := $order }}{{ if isset $done $k }}{{ $top = $i }}{{ end }}{{ end }} {{- $status := "Generally available" }} {{- if lt $top 2 }}{{ $status = printf "In %s" (index $order $top) }}{{ end }}
    {{- range $i, $k := $order }} {{- $name := index $names $k }} {{- with index $done $k }} {{- $date := partialCached "ledger/release-date.html" .release .release }}
  1. {{ partial "stage-mark.html" (dict "stage" $k) }}{{ $name }} v1.{{ .release }} {{ $date.Format "Jan 2006" }} {{- $links := slice }} {{- range $word, $paths := dict "Post" (.posts | default slice) "Talk" (.talks | default slice) }} {{- range $n, $path := $paths }} {{- with site.GetPage $path }} {{- $label := $word }} {{- if gt (len $paths) 1 }}{{ $label = printf "%s %d" $word (add $n 1) }}{{ end }} {{- $links = $links | append (printf `%s: %s` (partial "ledger/page-href.html" . | htmlEscape) $label (.Title | htmlEscape)) }} {{- else }} {{- errorf "data/ledger.yaml: %s %s: %q is not a page" $f.id $k $path }} {{- end }} {{- end }} {{- end }} {{- with $links }}{{ delimit . "" | safeHTML }}{{ end }}
    2. {{- else }} {{- if lt $i $top }}
  2. {{ $name }}Skipped
    3. {{- else }}
  3. {{ $name }}Not yet
    4. {{- end }} {{- end }} {{- end }}
