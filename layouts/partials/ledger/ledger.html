{{- /* The release ledger as a timing diagram. Releases are the clock; each feature is a signal that steps up a level (alpha low, beta mid, GA high) on the release that moved it. Everything is computed from data/ledger.yaml, so a new feature, stage or release needs no template change. Geometry: each release column is 100 units wide on a viewBox 300 units tall, drawn with preserveAspectRatio="none" and a non-scaling stroke. */ -}} {{- $k8s := hugo.Data.ledger.kubernetes }} {{- $first := int $k8s.first }} {{- $last := int $k8s.last }} {{- /* The release in development, set only while a target is drawn; else 0. */}} {{- $devData := int ($k8s.inDevelopment | default 0) }} {{- /* A released stage outside first..last, or a target anywhere but inDevelopment, means data/ledger.yaml is inconsistent. */}} {{- if and $devData (le $devData $last) }}{{ errorf "data/ledger.yaml: inDevelopment v1.%d is not after last v1.%d" $devData $last }}{{ end }} {{- $targets := 0 }} {{- range $k8s.features }}{{ $f := . }}{{ range .stages }} {{- if .target }} {{- $targets = add $targets 1 }} {{- if ne (int .release) $devData }}{{ errorf "data/ledger.yaml: %s %s target at v1.%d is not at inDevelopment" $f.id .stage .release }}{{ end }} {{- else if or (lt (int .release) $first) (gt (int .release) $last) }}{{ errorf "data/ledger.yaml: %s %s at v1.%d is outside v1.%d to v1.%d" $f.id .stage .release $first $last }}{{ end }} {{- end }}{{ end }} {{- if and $devData (not $targets) }}{{ errorf "data/ledger.yaml: inDevelopment is v1.%d but no stage is a target" $devData }}{{ end }} {{- /* Targets are drawn only while ledger/fresh.html vouches for them; otherwise the diagram shows released stages and ends at last. */}} {{- $fresh := partialCached "ledger/fresh.html" . }} {{- $dev := cond $fresh $devData 0 }} {{- $end := cond (gt $dev 0) $dev $last }} {{- /* At most $cap columns: past that, the oldest releases fold into the first column, labelled "≤N", and each feature shows only its latest stage from them there. */}} {{- $cap := 14 }} {{- $base := $first }} {{- if gt (add (sub $end $first) 1) $cap }}{{ $base = sub $end (sub $cap 1) }}{{ end }} {{- $folded := gt $base $first }} {{- $cols := add (sub $end $base) 1 }} {{- /* Released traces end at the right edge of last; the in-development column holds targets only. */}} {{- $relEnd := mul (add (sub $last $base) 1) 100 }} {{- $levels := dict "alpha" 195 "beta" 110 "ga" 25 }} {{- $glyphs := dict "alpha" 1 "beta" 2 "ga" 3 }} {{- $width := mul $cols 100 }} {{- /* Short stage-link label per post source; unknown sources show their own name. */}} {{- $sourceLabels := dict "Kubernetes.io" "k8s.io" "Microsoft Open Source" "Microsoft" }}
