{{- /* Map of content path -> ledger references, built from data/ledger.yaml, in ledger order. Each reference: id, name, kep, stage, release, kind (feature or subproject). Targets are left out: a post or talk is tagged only with a stage that has shipped. */ -}} {{- $refs := dict }} {{- range $f := hugo.Data.ledger.kubernetes.features }} {{- range $s := $f.stages }} {{- if $s.target }}{{ continue }}{{ end }} {{- $ref := dict "id" $f.id "name" $f.name "kep" $f.kep "stage" $s.stage "release" $s.release "kind" "feature" }} {{- range $path := (append ($s.posts | default slice) ($s.talks | default slice)) }} {{- $refs = merge $refs (dict $path (append (index $refs $path | default slice) (slice $ref))) }} {{- end }} {{- end }} {{- end }} {{- range $sp := hugo.Data.ledger.subprojects }} {{- $ref := dict "id" $sp.id "name" $sp.name "kep" $sp.kep "stage" $sp.stage "release" $sp.release "kind" "subproject" }} {{- range $path := ($sp.talks | default slice) }} {{- $refs = merge $refs (dict $path (append (index $refs $path | default slice) (slice $ref))) }} {{- end }} {{- end }} {{- return $refs }}
